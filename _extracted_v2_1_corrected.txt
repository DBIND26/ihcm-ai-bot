2. Auth and Middleware — Corrected
BUG: V2.0 middleware matcher was '/(platform)/:path*'. Next.js route-group names like (platform) are NOT part of the URL. The matcher would never fire. /dashboard, /ask, etc. would be unprotected.
FIX: Matcher now lists each top-level route explicitly. Role is read from profiles and injected into request headers so API routes can check it without an extra DB call.
2.1  Corrected middleware.js
// middleware.js — CORRECTED V2.1
import { createMiddlewareClient } from '@supabase/auth-helpers-nextjs'
import { NextResponse } from 'next/server'
const PROTECTED_ROUTES = [
'/dashboard', '/ask', '/workflows',
'/building',  '/knowledge', '/settings'
]
export async function middleware(req) {
const { pathname } = req.nextUrl
const isProtected = PROTECTED_ROUTES.some(r => pathname.startsWith(r))
if (!isProtected) return NextResponse.next()
const res = NextResponse.next()
const supabase = createMiddlewareClient({ req, res })
const { data: { session } } = await supabase.auth.getSession()
if (!session) {
return NextResponse.redirect(new URL('/login', req.url))
}
// Fetch role and inject into header for API routes
const { data: profile } = await supabase
.from('profiles')
.select('role, primary_building_id')
.eq('id', session.user.id)
.single()
if (!profile || profile.role === 'pending') {
return NextResponse.redirect(new URL('/pending', req.url))
}
const requestHeaders = new Headers(req.headers)
requestHeaders.set('x-user-role', profile.role)
requestHeaders.set('x-user-id', session.user.id)
requestHeaders.set('x-building-id', profile.primary_building_id || '')
return NextResponse.next({ request: { headers: requestHeaders } })
}

-- CORRECTED VIEW: JOINs buildings to expose name/cms/state/strategic_priority
-- Also derives census_trend from last two metric rows
create view v_latest_building_metrics as
select
bm.*,
b.name,
b.short_name,
b.cms_provider_id,
b.state,
b.market,
b.strategic_priority,
-- derive census_trend by comparing current row to prior row
case
when bm.census_current > lag(bm.census_current) over (
partition by bm.building_id order by bm.metric_date) then 'growing'
when bm.census_current < lag(bm.census_current) over (
partition by bm.building_id order by bm.metric_date) then 'declining'
else 'flat'
end as census_trend
from (
select distinct on (building_id) *
from building_metrics
order by building_id, metric_date desc
) bm
join buildings b on b.id = bm.building_id;

6.2  Corrected retrieveKnowledge()
// lib/knowledge/retrieve.js — CORRECTED V2.1
// Staleness thresholds in days by source_type
const STALE_DAYS = {
building_profile:        14,
payer_guidance:          45,
state_reimbursement:     90,
survey_history:          90,
poc_template:            90,
operator_best_practice:  180,
corporate_playbook:      180,
approved_faq:            90,
policy:                  365,
referral_intelligence:   30,
}
export async function retrieveKnowledge(query, opts = {}) {
const { buildingId, state, sourceTypes, limit = 5 } = opts
let q = supabase
.from('knowledge_sources')
.select(`
id, title, citation_text, effective_date, full_content,
source_type, review_due_date, status, storage_path,
owner:owner_id ( full_name ),
ts_rank_cd(to_tsvector('english', coalesce(title,'') || ' ' || coalesce(full_content,'')),
plainto_tsquery('english', '${query}')) as relevance_score
`)
.eq('status', 'approved')
.textSearch('full_content', query, { type: 'websearch' })
.order('relevance_score', { ascending: false })
.limit(limit)
if (buildingId) q = q.or(`building_id.eq.${buildingId},building_id.is.null`)
if (state)      q = q.or(`state.eq.${state},state.is.null`)
if (sourceTypes) q = q.in('source_type', sourceTypes)
const { data } = await q
if (!data) return []
const now = Date.now()
return data.map(src => {
const threshold = STALE_DAYS[src.source_type] ?? 90
const ageDays = src.review_due_date
? Math.floor((now - new Date(src.review_due_date)) / 86400000)
: null
return {
...src,
owner_name: src.owner?.full_name ?? 'Unassigned',
is_stale: ageDays !== null && ageDays > 0,
stale_days_overdue: ageDays !== null && ageDays > 0 ? ageDays : 0,
}
})
}
